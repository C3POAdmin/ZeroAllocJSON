cmake_minimum_required(VERSION 3.10)
project(ZeroAllocJSON VERSION 0.0.1 LANGUAGES CXX)
add_library(ZeroAllocJSON src/ZeroAllocJSON.cpp)
target_include_directories(ZeroAllocJSON PUBLIC src)
set_target_properties(ZeroAllocJSON PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED YES CXX_EXTENSIONS NO)
option(ZEROALLOCJSON_TESTS "Build POSIX host tests" ON)
if(ZEROALLOCJSON_TESTS)
  enable_testing()
  add_executable(zeroallocjson_tests test/test.cpp)
  target_link_libraries(zeroallocjson_tests PRIVATE ZeroAllocJSON)
  target_compile_options(zeroallocjson_tests PRIVATE -UNDEBUG)
  add_test(NAME zeroallocjson COMMAND zeroallocjson_tests)
endif()
